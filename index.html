nameMatch[1].trim() : "Canal sin nombre"; const logo = logoMatch ? logoMatch[1].trim() : ""; const group = groupMatch ? groupMatch[1].trim() : "Sin grupo"; const url = lines[i + 1]?.trim(); if (url && url.startsWith("http")) { if (!seenGroups.has(group)) { groupOrder.push(group); seenGroups.add(group); } channels.push({ name, logo, group, url }); } } } return channels; } function renderFiltered() { const query = document.getElementById("search").value.toLowerCase(); const filtered = allChannels.filter(c => c.name.toLowerCase().includes(query)); renderGrouped(filtered); } function renderGrouped(channels) { const container = document.getElementById("channels"); container.innerHTML = ""; const grouped = {}; groupOrder.forEach(g => grouped[g] = []); channels.forEach(ch => { if (!grouped[ch.group]) grouped[ch.group] = []; grouped[ch.group].push(ch); }); groupOrder.forEach(group => { if (grouped[group].length === 0) return; const title = document.createElement("div"); title.className = "group"; title.textContent = group; const groupGrid = document.createElement("div"); groupGrid.className = "group-channels"; grouped[group].forEach(ch => { const div = document.createElement("div"); div.className = "channel"; div.innerHTML = \`
